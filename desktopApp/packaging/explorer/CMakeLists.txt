cmake_minimum_required(VERSION 3.24)
project(GameTierExplorer LANGUAGES CXX)

if(NOT WIN32)
    message(FATAL_ERROR "The Explorer command requires Windows")
endif()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
if(MSVC)
    add_compile_options(/W4 /permissive- /EHsc /utf-8)
else()
    add_compile_options(-Wall -Wextra -Wpedantic)
endif()

add_library(GameTierExplorer SHARED src/ExplorerCommand.cpp src/ManagedSelection.h src/GameTierExplorer.def)
set_target_properties(GameTierExplorer PROPERTIES PREFIX "")
target_link_libraries(GameTierExplorer PRIVATE shell32 shlwapi pathcch ole32 advapi32 uuid)
install(TARGETS GameTierExplorer RUNTIME DESTINATION .)
install(FILES third_party/nlohmann/LICENSE.MIT DESTINATION . RENAME GameTierExplorer.LICENSE.txt)

include(CTest)
if(BUILD_TESTING)
    add_executable(GameTierExplorerTest tests/ExplorerCommandTest.cpp)
    target_link_libraries(GameTierExplorerTest PRIVATE shell32 ole32 advapi32 uuid)
    add_executable(GameTierLaunchRecorder tests/LaunchRecorder.cpp)
    set_target_properties(GameTierLaunchRecorder PROPERTIES OUTPUT_NAME GameTier)
    if(MINGW)
        target_link_options(GameTierExplorerTest PRIVATE -municode)
        target_link_options(GameTierLaunchRecorder PRIVATE -municode)
    endif()
    add_dependencies(GameTierExplorerTest GameTierExplorer GameTierLaunchRecorder)
    add_test(NAME ExplorerCommandTest COMMAND GameTierExplorerTest $<TARGET_FILE:GameTierExplorer>)
    add_executable(GameTierPathIdentityTest tests/PathIdentityTest.cpp)
    target_link_libraries(GameTierPathIdentityTest PRIVATE shell32 shlwapi pathcch ole32 advapi32 uuid)
    add_test(NAME PathIdentityTest COMMAND GameTierPathIdentityTest)
endif()
